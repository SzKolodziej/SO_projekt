#!/bin/bash

alph=(.- -... -.-. -.. . ..-. --. .... .. .--- -.- .-.. -- -. --- .--. --.- .-. ... - ..- ...- .-- -..- -.-- --..)
num=(----- .---- ..--- ...-- ....- ..... -.... --... ---.. ----.)

result=()

dot="."
dash="-"

help() {
	echo "help section"
	exit 0
}

if [[ $# == 0 ]]; then
	help
fi

while [[ $# -gt 0 ]]
do
	case "$1" in
		-h|--help)
			help
			;;
		-d=*)
			dot="${1#-d=}"
			shift
			;;
		-D=*)
			dash="${1#-D=}"
			shift
			;;
		--)
			shift
			break
			;;
		-*)
			echo "Unknown option: $1"
			exit 1
			;;
		*)
			break
			;;
	esac
done

text="$*"

if [[ -z $text ]]; then
	echo "error: text is missing"
	exit 1
fi

for ((i = 0; i < ${#text}; i++))
do
	char=$(printf '%d' "'${text:i:1}")

	if [[ $char -ge 65 ]] && [[ $char -le 90 ]]; then
		result+="${alph[$(($char % 65))]} "
	elif [[ $char -ge 97 ]] && [[ $char -le 122 ]]; then
		result+="${alph[$(($char % 97))]} "
	elif [[ $char -ge 48 ]] &&  [[ $char -le 57 ]]; then
		result+="${num[$(($char % 48))]} "
	elif [[ $char == 32 ]]; then
		#echo -n " "
		result+="/ "
	else
		result="Invalid symbol"
		break
	fi
done

echo $result|tr .- $dot$dash
